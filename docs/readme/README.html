def rebuild() -> int:
    chunks = policy_chunks() + product_chunks()          # 안내 12 + 상품 30
    vectors = embed([c["content"] for c in chunks])      # text-embedding-3-small
    db.table("rag_chunks").insert([{**c, "embedding": v} for c, v in zip(chunks, vectors)])

def search(query, source=None, k=5, vector=None):
    vec = vector or embed([query])[0]
    return get_client().rpc("match_rag_chunks", {"query_embedding": vec,
        "match_count": k, "filter_source": source}).execute().data

create function match_rag_chunks(query_embedding vector(1536), match_count int, filter_source text)
returns table (id bigint, title text, content text, similarity float) as $$
  select id, title, content, 1 - (embedding <=> query_embedding) as similarity
  from rag_chunks where filter_source is null or source = filter_source
  order by embedding <=> query_embedding limit match_count;
$$;

def photo_vector(img) -> np.ndarray:
    """사진 → YOLO 분류 모델의 특징 1280개 (길이 1로 맞춤)"""
    x = center_crop(img, 224) / 255.0
    v = _load(FEATURE_FILE).run(None, {"images": x})[0][0]
    return v / np.linalg.norm(v)

def match_product(img):
    ids, matrix = _product_vectors()          # 상품 30개 사진 특징 (5분 캐시)
    scores = matrix @ photo_vector(img)        # 코사인 유사도
    best = int(scores.argmax())
    if scores[best] < SAME_PRODUCT:            # 0.75 미만이면 같은 상품 아님
        return None
    return {"product_id": ids[best], "similarity": round(float(scores[best]), 3)}
